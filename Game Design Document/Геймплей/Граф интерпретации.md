## Структура графа

Граф состоит из трех уровней узлов, соединенных направленными связями.

**Нижний уровень — подсказки.** Шесть узлов, по два на каждую улику. Каждая пара содержит одну наивную и одну циничную подсказку. Подсказки обнаруживаются в ходе исследования локации и изначально скрыты.

**Средний уровень — улики.** Три узла. Каждая улика связана ровно с двумя подсказками — одной каждого типа. Улика имеет три возможных состояния: нераскрытая, интерпретирована наивно, интерпретирована цинично.

**Верхний уровень — центральный вопрос главы.** Один узел. Его интерпретация определяется совокупностью трактовок всех трех улик и разрешается при завершении главы.

---

## Подсказки

Подсказки находятся независимо друг от друга в произвольном порядке. Нахождение подсказки не дает трактовки — оно лишь открывает возможность выбора.

Каждая подсказка принадлежит ровно одной улике и ровно одному типу интерпретации. Связь подсказки с уликой визуально обозначается линией соответствующего цвета: зеленой для наивной, красной для циничной.

---

## Улики

Улика остается нераскрытой, пока для нее не найдена хотя бы одна подсказка.

Как только найдена первая подсказка улики, улика автоматически получает ее тип интерпретации. Если впоследствии найдена вторая подсказка того же артефакта (противоположного типа), игрок получает возможность переключить трактовку вручную.

Переключение между наивной и циничной интерпретацией доступно только при наличии обеих подсказок для данной улики. При наличии лишь одной подсказки трактовка фиксирована.

---

## Связанные улики

Две из трех улик могут быть связаны между собой тематически или сюжетно. Связь работает следующим образом:

- при выборе трактовки одной из связанных улик вторая автоматически получает согласованную трактовку, если та еще не была раскрыта
- если обе улики уже раскрыты и их трактовки расходятся, возникает состояние **противоречия** — оба узла визуально выделяются, а в итоговом вердикте отображается предупреждение
- противоречие не блокирует завершение главы, но фиксируется в результате

Третья улика связей не имеет и интерпретируется независимо.

---

## Завершение главы

Глава завершается вручную — игрок инициирует подведение итогов самостоятельно.

Перед завершением игрок указывает свой ответ на центральный вопрос главы: наивный или циничный. Этот выбор применяется ко всем уликам, для которых не было найдено ни одной подсказки и которые остались нераскрытыми.

Итоговая интерпретация центрального вопроса определяется большинством:

|Соотношение трактовок|Интерпретация главы|
|---|---|
|Наивных больше|Наивная|
|Циничных больше|Циничная|
|Равное количество|Нейтральная|

    line.setAttribute("y2", r1.top-br.top+r1.height/2);

    line.setAttribute("stroke","#3a2e22");

    line.setAttribute("stroke-width","2");

    line.setAttribute("stroke-dasharray","7,4");

    line.id='link-line';

    svg.appendChild(line);

  }

  

  refreshUI();

}

  

function onClueClick(id){

  found[id]=!found[id];

  document.getElementById(id).classList.toggle('found',found[id]);

  autoUpdateArtifacts();

  refreshUI();

}

  

function autoUpdateArtifacts(){

  for(let i=0;i<numArtifacts;i++){

    const hasN=CLUES.some(c=>c.artifact===i&&c.alignment==='naive'&&found[c.id]);

    const hasC=CLUES.some(c=>c.artifact===i&&c.alignment==='cynic'&&found[c.id]);

    const cur=artifactState[i];

    if(!hasN&&!hasC){

      if(cur!=='unknown'){

        artifactState[i]='unknown';

        if(LINKED_ARTIFACTS.includes(i)){

          const li=LINKED_ARTIFACTS.find(x=>x!==i);

          const lN=CLUES.some(c=>c.artifact===li&&c.alignment==='naive'&&found[c.id]);

          const lC=CLUES.some(c=>c.artifact===li&&c.alignment==='cynic'&&found[c.id]);

          if(!lN&&!lC) artifactState[li]='unknown';

        }

      }

    } else if(cur==='naive'&&!hasN){

      if(hasC){

        artifactState[i]='cynic';

        if(LINKED_ARTIFACTS.includes(i)){

          const li=LINKED_ARTIFACTS.find(x=>x!==i);

          if(artifactState[li]!=='unknown') artifactState[li]='cynic';

        }

      }

    } else if(cur==='cynic'&&!hasC){

      if(hasN){

        artifactState[i]='naive';

        if(LINKED_ARTIFACTS.includes(i)){

          const li=LINKED_ARTIFACTS.find(x=>x!==i);

          if(artifactState[li]!=='unknown') artifactState[li]='naive';

        }

      }

    }

  }

}

  

function countSelected(){ return Object.values(artifactState).filter(s=>s!=='unknown').length; }

  

function areLinkedConflicted(){

  const s0=artifactState[LINKED_ARTIFACTS[0]];

  const s1=artifactState[LINKED_ARTIFACTS[1]];

  if(s0==='unknown'||s1==='unknown') return false;

  return s0!==s1;

}

  

function onArtifactClick(i){

  const hasN=CLUES.some(c=>c.artifact===i&&c.alignment==='naive'&&found[c.id]);

  const hasC=CLUES.some(c=>c.artifact===i&&c.alignment==='cynic'&&found[c.id]);

  const cur=artifactState[i];

  if(cur==='unknown'){

    if(hasN){ artifactState[i]='naive'; applyLink(i,'naive'); }

    else if(hasC){ artifactState[i]='cynic'; applyLink(i,'cynic'); }

    else return;

  } else if(cur==='naive'){

    if(hasC){ artifactState[i]='cynic'; applyLink(i,'cynic'); }

  } else {

    if(hasN){ artifactState[i]='naive'; applyLink(i,'naive'); }

  }

  refreshUI();

}

  

function applyLink(i, val){

  if(!LINKED_ARTIFACTS.includes(i)) return;

  const li=LINKED_ARTIFACTS.find(x=>x!==i);

  if(artifactState[li]==='unknown') artifactState[li]=val;

  else if(artifactState[li]!=='unknown') artifactState[li]=val;

}

  });

  if(!svg) svg=createSVG();

  const linkLine=svg.querySelector('#link-line');

  svg.innerHTML='';

  const br=board.getBoundingClientRect();

  

  // Clue → artifact lines

  CLUES.forEach(c=>{

    if(!found[c.id]) return;

    const ce=document.getElementById(c.id).getBoundingClientRect();

    const ae=document.getElementById('artifact-'+c.artifact).getBoundingClientRect();

    const line=document.createElementNS("http://www.w3.org/2000/svg","line");

    line.setAttribute("stroke-width","1.5");

    svg.appendChild(line);

  });

  

  let counts={naive:0,cynic:0,unknown:0};

  const isConflicted=areLinkedConflicted();

  

  for(let i=0;i<numArtifacts;i++){

    const el=document.getElementById('artifact-'+i);

    el.classList.remove('unknown','naive','cynic','conflicted');

    el.classList.add(artifactState[i]);

    if(isConflicted&&LINKED_ARTIFACTS.includes(i)) el.classList.add('conflicted');

  

    const stEl=document.getElementById('art-'+i+'-state');

    stEl.textContent=

      artifactState[i]==='unknown'?'нераскрыта':

      artifactState[i]==='naive'?'наивная':'циничная';

      line.setAttribute("y2", me.top-br.top+me.height/2);

      line.setAttribute("stroke", artifactState[i]==='naive'?'rgba(39,174,96,0.35)':'rgba(192,57,43,0.35)');

      line.setAttribute("stroke-width","1.5");

      svg.appendChild(line);

    }

  }

  

  // Main node verdict

  let mainVerdict='—';

  if(chapterFinished){

    if(counts.naive>counts.cynic) mainVerdict='Наивная';

    else if(counts.cynic>counts.naive) mainVerdict='Циничная';

    else mainVerdict='Нейтральная';

  }

  document.getElementById('mainState').textContent=mainVerdict;

  

  // Verdict panel

  const vEl=document.getElementById('verdict');

  vEl.className='verdict-text';

  if(chapterFinished){

    vEl.textContent=mainVerdict+(isConflicted?' — противоречие':'');

    if(counts.naive>counts.cynic) vEl.classList.add('naive');

    else if(counts.cynic>counts.naive) vEl.classList.add('cynic');

    if(isConflicted){

      vEl.innerHTML+='<span class="conflict-badge">Конфликт</span>';

    }

  } else {

    vEl.textContent='Глава не завершена';

  }

  

  document.getElementById('verDetails').innerHTML=

    `<span style="color:var(--naive-col)">Наивных: ${counts.naive}</span> &nbsp;`+

    `<span style="color:var(--cynic-col)">Циничных: ${counts.cynic}</span> &nbsp;`+

    `<span>Нераскрытых: ${counts.unknown}</span>`+

    (isConflicted?`<div class="warning">⚠ Улики 1 и 2 противоречат друг другу</div>`:'');

  

  // Found clues list

  const foundList=document.getElementById('foundList');

  const items=CLUES.filter(c=>found[c.id]);

  foundList.innerHTML=items.length

    ? items.map(f=>`<div style="margin-bottom:3px;color:${f.alignment==='naive'?'var(--naive-col)':'var(--cynic-col)'}">· ${f.title}</div>`).join('')

    : '—';

  

  // Art list

  const artList=document.getElementById('artList');

  artList.innerHTML='';

  for(let i=0;i<numArtifacts;i++){

    const hasN=CLUES.some(c=>c.artifact===i&&c.alignment==='naive'&&found[c.id]);

    const hasC=CLUES.some(c=>c.artifact===i&&c.alignment==='cynic'&&found[c.id]);

    const row=document.createElement('div');

    row.className='art-row';

    const stateColor=

      artifactState[i]==='naive'?'var(--naive-col)':

      artifactState[i]==='cynic'?'var(--cynic-col)':'var(--text)';

  

    row.innerHTML=`<div class="small">

      <strong style="font-family:'Cinzel',serif;font-size:9px;letter-spacing:0.15em;color:${stateColor}">

        ${linked?'↔ ':''}Улика ${i+1}

      </strong>

      <div style="font-size:10px;margin-top:2px">${ARTIFACT_TITLES[i]}</div>

    </div>`;

  

    if(hasN&&hasC&&artifactState[i]!=='unknown'){

      const btn=document.createElement('button');

      btn.textContent='Переключить';

      btn.style.fontSize='8px';

      btn.style.padding='5px 8px';

      btn.onclick=()=>onArtifactClick(i);

      row.appendChild(btn);

    }

    artList.appendChild(row);

  }

}
